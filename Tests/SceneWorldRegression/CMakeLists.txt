cmake_minimum_required(VERSION 3.14)
project(SceneWorldRegression LANGUAGES CXX)
enable_testing()
set(TOMCAT_EKIT_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../TomCat/vendor/ekit/include" CACHE PATH "ekit headers to validate")
add_executable(SceneWorldRegression src/SceneWorldRegression.cpp)
target_compile_features(SceneWorldRegression PRIVATE cxx_std_20)
target_include_directories(SceneWorldRegression PRIVATE ../../TomCat/src "${TOMCAT_EKIT_INCLUDE_DIR}")
add_test(NAME SceneWorldRegression COMMAND SceneWorldRegression)
option(TOMCAT_BUILD_SCENE_WORLD_BENCHMARK "Build the View/ForEach comparison" OFF)
if(TOMCAT_BUILD_SCENE_WORLD_BENCHMARK)
    add_executable(SceneWorldBenchmark benchmark.cpp)
    target_compile_features(SceneWorldBenchmark PRIVATE cxx_std_20)
    target_include_directories(SceneWorldBenchmark PRIVATE ../../TomCat/src "${TOMCAT_EKIT_INCLUDE_DIR}")
endif()
